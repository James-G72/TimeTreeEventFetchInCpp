cmake_minimum_required(VERSION 3.10)
project(TimeTreeHandlerCpp LANGUAGES CXX)
find_package(CURL REQUIRED)
add_library(timetree STATIC timetree_utils.cpp timetree_structs.cpp)
target_compile_features(timetree PUBLIC cxx_std_17)
target_include_directories(timetree PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(TARGET CURL::libcurl)
    target_link_libraries(timetree PUBLIC CURL::libcurl)
else()
    # FindCURL gained its imported target after the locally installed CMake 3.10.
    target_include_directories(timetree PRIVATE ${CURL_INCLUDE_DIRS})
    target_link_libraries(timetree PUBLIC ${CURL_LIBRARIES})
endif()
option(TIMETREE_BUILD_EXAMPLE "Build the command-line query example" OFF)
if(TIMETREE_BUILD_EXAMPLE)
    add_executable(timetree_example main.cpp)
    target_link_libraries(timetree_example PRIVATE timetree)
endif()
