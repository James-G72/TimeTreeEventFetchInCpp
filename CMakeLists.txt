cmake_minimum_required(VERSION 3.10)
project(TimeTreeHandlerCpp LANGUAGES CXX)
add_library(timetree STATIC timetree_utils.cpp timetree_structs.cpp)
target_compile_features(timetree PUBLIC cxx_std_17)
target_include_directories(timetree PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

option(TIMETREE_BUILD_EXAMPLE "Build the desktop HTTP client and live viewer" OFF)
if(TIMETREE_BUILD_EXAMPLE)
    find_package(CURL REQUIRED)
    add_executable(timetree_example main.cpp)
    target_link_libraries(timetree_example PRIVATE timetree)
    if(TARGET CURL::libcurl)
        target_link_libraries(timetree_example PRIVATE CURL::libcurl)
    else()
        target_include_directories(timetree_example PRIVATE ${CURL_INCLUDE_DIRS})
        target_link_libraries(timetree_example PRIVATE ${CURL_LIBRARIES})
    endif()
endif()

option(TIMETREE_BUILD_TESTS "Build the offline library tests" OFF)
if(TIMETREE_BUILD_TESTS)
    enable_testing()
    foreach(test_name integration recurrence)
        add_executable(timetree_${test_name} tests/${test_name}.cpp)
        target_link_libraries(timetree_${test_name} PRIVATE timetree)
        add_test(NAME ${test_name} COMMAND timetree_${test_name})
    endforeach()
endif()
